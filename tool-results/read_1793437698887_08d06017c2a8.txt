    85→    85→function describeEvent(ev: ActivityEvent): string {
    86→    86→  const u = `@${ev.user?.uniqueId || ev.user?.nickname || 'viewer'}`;
    87→    87→  switch (ev.type) {
    88→    88→    case 'claim': return `${u} locked slot #${ev.slot}`;
    89→    89→    case 'rejected': return `${u} was too late — #${ev.slot} already taken`;
    90→    90→    case 'duplicate': return `${u} already holds slot #${ev.slot}`;
    91→    91→    case 'invalid': return `${u} typed "${ev.comment || '?'}" — not a valid slot`;
    92→    92→    case 'gift': return `${u} sent ${ev.giftName || 'a gift'} ×${ev.repeatCount || 1}`;
    93→    93→    case 'join': return `${u} joined the stream`;
    94→    94→    case 'released': return `Slot #${ev.slot} was released`;
    95→    95→    case 'grid-full': return `Grid complete! Round ${ev.round} — auto-reset in 6s`;
    96→    96→    case 'reset': return `Round ${ev.round} started — grid is open`;
    97→    97→    case 'chat': return `${u}: "${ev.comment || ''}"`;
    98→    98→    default: return ev.type;
    99→    99→  }
   100→   100→}
