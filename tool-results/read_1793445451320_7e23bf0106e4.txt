    96→function describeEvent(ev: ActivityEvent): string {
    97→  const u = `@${ev.user?.uniqueId || ev.user?.nickname || 'viewer'}`;
    98→  switch (ev.type) {
    99→    case 'claim': return `${u} locked slot #${ev.slot}`;
   100→    case 'rejected': return `${u} was too late — #${ev.slot} already taken`;
   101→    case 'duplicate': return `${u} already holds slot #${ev.slot}`;
   102→    case 'invalid': return `${u} typed "${ev.comment || '?'}" — not a valid slot`;
   103→    case 'gift': return `${u} sent ${ev.giftName || 'a gift'} ×${ev.repeatCount || 1}`;
   104→    case 'join': return `${u} joined the stream`;
   105→    case 'released': return `Slot #${ev.slot} was released`;
   106→    case 'grid-full': return `Grid complete — the lucky number is coming up!`;
   107→    case 'busy': return `${u} typed ${ev.comment ?? 'a number'} during the draw — round closed`;
   108→    case 'draw-started': return `Keno draw started — ${(ev as { poolSize?: number }).poolSize ?? '?'} slots in the pool`;
   109→    case 'draw-result': return `👑 @${(ev as { winner?: { uniqueId?: string } }).winner?.uniqueId || 'winner'} won slot #${ev.slot} — round ${ev.round} complete!`;
   110→    case 'reset': return `Round ${ev.round} started — grid is open`;
   111→    case 'chat': return `${u}: "${ev.comment || ''}"`;
   112→    default: return ev.type;
   113→  }
   114→}
