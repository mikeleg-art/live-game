    82→    82→function describeEvent(ev: ActivityEvent): string {
    83→    83→  const u = `@${ev.user?.uniqueId || ev.user?.nickname || 'viewer'}`;
    84→    84→  switch (ev.type) {
    85→    85→    case 'claim': return `${u} locked slot #${ev.slot}`;
    86→    86→    case 'rejected': return `${u} was too late — #${ev.slot} already taken`;
    87→    87→    case 'duplicate': return `${u} already holds slot #${ev.slot}`;
    88→    88→    case 'invalid': return `${u} typed "${ev.comment || '?'}" — not a valid slot`;
    89→    89→    case 'gift': return `${u} sent ${ev.giftName || 'a gift'} ×${ev.repeatCount || 1}`;
    90→    90→    case 'join': return `${u} joined the stream`;
    91→    91→    case 'released': return `Slot #${ev.slot} was released`;
    92→    92→    case 'grid-full': return `Grid complete! Round ${ev.round} — auto-reset in 6s`;
    93→    93→    case 'reset': return `Round ${ev.round} started — grid is open`;
    94→    94→    case 'chat': return `${u}: "${ev.comment || ''}"`;
    95→    95→    default: return ev.type;
    96→    96→  }
    97→    97→}
